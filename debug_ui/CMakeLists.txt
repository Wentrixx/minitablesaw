cmake_minimum_required(VERSION 3.20)
project(minitablesaw_debug LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT WIN32)
    message(FATAL_ERROR "This serial backend requires Windows")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
add_executable(minitablesaw_debug main.c window.c serial.c)
get_filename_component(FIRMWARE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
target_compile_definitions(minitablesaw_debug PRIVATE FIRMWARE_ROOT="${FIRMWARE_ROOT}")
target_link_libraries(minitablesaw_debug PRIVATE PkgConfig::GTK4)
target_compile_options(minitablesaw_debug PRIVATE -Wall -Wextra -Wpedantic)
